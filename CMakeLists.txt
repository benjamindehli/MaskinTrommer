# maskintrommer-plugin — distributable product (VST3 / AU / Standalone).
# Thin wrapper around dehli_musikk_sampler_engine (dm::ManifestPluginProcessor).

juce_add_plugin(MaskinTrommer
    COMPANY_NAME            "DehliMusikk"
    BUNDLE_ID               com.dehlimusikk.maskintrommer
    VERSION                 0.1.0
    PLUGIN_MANUFACTURER_CODE Dhmk
    PLUGIN_CODE              Mskt
    FORMATS                 VST3 AU Standalone
    PRODUCT_NAME            "MaskinTrommer"
    IS_SYNTH                TRUE
    NEEDS_MIDI_INPUT        TRUE
    NEEDS_MIDI_OUTPUT       FALSE
    IS_MIDI_EFFECT          FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD TRUE
    VST3_CATEGORIES         "Instrument" "Sampler"
    AU_MAIN_TYPE            "kAudioUnitType_MusicDevice"
)

target_sources(MaskinTrommer PRIVATE
    source/Plugin.cpp
)

target_compile_definitions(MaskinTrommer PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(MaskinTrommer PRIVATE
    dehli_musikk_sampler_engine
    juce::juce_audio_utils
    juce::juce_recommended_config_flags
    juce::juce_recommended_warning_flags
)

# Embedded asset bundle (generated by ds-plugin-converter into assets/).
file(GLOB MASKIN_ASSETS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/manifest.json"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/samples/*.flac"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/ir/*.flac"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/images/*.png"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/images/*.jpg"
    "${CMAKE_CURRENT_SOURCE_DIR}/assets/images/*.jpeg")

if(MASKIN_ASSETS)
    juce_add_binary_data(MaskinTrommerAssets SOURCES ${MASKIN_ASSETS})
    target_link_libraries(MaskinTrommer PRIVATE MaskinTrommerAssets)
    target_compile_definitions(MaskinTrommer PRIVATE MASKINTROMMER_HAS_ASSETS=1)
    list(LENGTH MASKIN_ASSETS MASKIN_ASSET_COUNT)
    message(STATUS "MaskinTrommer: embedding ${MASKIN_ASSET_COUNT} asset file(s) from assets/")
else()
    message(WARNING "MaskinTrommer: no assets in assets/ — plugin will be silent. "
                    "Generate them with the dmse_convert target first.")
endif()
